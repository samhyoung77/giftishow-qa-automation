## 📝 POM 패턴 예시 코드

### BasePage (base_page.py)

```python
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.common.by import By

class BasePage:
    def __init__(self, driver):
        self.driver = driver
        self.wait = WebDriverWait(driver, 10)

    def find_element(self, locator):
        return self.wait.until(EC.presence_of_element_located(locator))

    def click(self, locator):
        element = self.find_element(locator)
        element.click()

    def input_text(self, locator, text):
        element = self.find_element(locator)
        element.clear()
        element.send_keys(text)

    def get_text(self, locator):
        return self.find_element(locator).text

    def take_screenshot(self, name):
        self.driver.save_screenshot(f"reports/screenshots/{name}.png")
```

### LoginPage (login_page.py)

```python
from selenium.webdriver.common.by import By
from pages.base_page import BasePage

class LoginPage(BasePage):
    # Locators
    USERNAME_INPUT = (By.ID, "username")
    PASSWORD_INPUT = (By.ID, "password")
    LOGIN_BUTTON = (By.CSS_SELECTOR, "button[type='submit']")
    ERROR_MESSAGE = (By.CLASS_NAME, "error-message")

    def __init__(self, driver):
        super().__init__(driver)
        self.url = "https://biz.giftishow.com/login"

    def navigate(self):
        self.driver.get(self.url)

    def login(self, username, password):
        self.input_text(self.USERNAME_INPUT, username)
        self.input_text(self.PASSWORD_INPUT, password)
        self.click(self.LOGIN_BUTTON)

    def get_error_message(self):
        return self.get_text(self.ERROR_MESSAGE)

    def is_login_successful(self):
        # 로그인 후 URL 변경 확인
        return "main" in self.driver.current_url
```

### Test Case (test_login.py)

```python
import pytest
from pages.login_page import LoginPage
from utils.test_reporter import TestReporter

class TestLogin:
    @pytest.fixture(autouse=True)
    def setup(self, driver):
        self.login_page = LoginPage(driver)
        self.reporter = TestReporter()
        yield

    def test_successful_login(self, driver, test_data):
        """TC_GS_LOGIN_001: 정상 로그인 테스트"""
        # Arrange
        self.login_page.navigate()

        # Act
        self.login_page.login(
            test_data['valid_user']['username'],
            test_data['valid_user']['password']
        )

        # Assert
        assert self.login_page.is_login_successful()

        # Report to Google Sheets
        self.reporter.log_result(
            tc_id="GS_LOGIN_001",
            page="Login Page",
            scenario="정상 로그인",
            result="PASS"
        )
```

### Google Sheets Reporter (google_sheets.py)

```python
import gspread
from oauth2client.service_account import ServiceAccountCredentials
from datetime import datetime

class GoogleSheetsReporter:
    def __init__(self, sheet_url):
        scope = ['https://spreadsheets.google.com/feeds',
                 'https://www.googleapis.com/auth/drive']
        creds = ServiceAccountCredentials.from_json_keyfile_name(
            'credentials.json', scope)
        self.client = gspread.authorize(creds)
        self.sheet = self.client.open_by_url(sheet_url)

    def log_test_result(self, tc_id, page, scenario, result,
                       duration=0, error_msg="", screenshot_url=""):
        worksheet = self.sheet.worksheet("TestResults")

        row = [
            datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            tc_id,
            page,
            scenario,
            "Chrome",  # Browser
            result,
            duration,
            error_msg,
            screenshot_url
        ]

        worksheet.append_row(row)

    def update_summary(self):
        results_sheet = self.sheet.worksheet("TestResults")
        summary_sheet = self.sheet.worksheet("TestSummary")

        # 전체 결과 집계
        all_results = results_sheet.get_all_records()
        total = len(all_results)
        pass_count = sum(1 for r in all_results if r['Result'] == 'PASS')
        fail_count = total - pass_count
        pass_rate = (pass_count / total * 100) if total > 0 else 0

        # Summary 시트 업데이트
        summary_sheet.update('A2:E2', [[
            datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            total,
            pass_count,
            fail_count,
            f"{pass_rate:.1f}%"
        ]])
```

## 🔗 Google Sheets API 인증 설정

### 1. Google Cloud Console 설정
1. https://console.cloud.google.com 접속
2. 새 프로젝트 생성 (예: "Giftishow-Test")
3. API 및 서비스 → 라이브러리
4. "Google Sheets API" 검색 후 사용 설정
5. "Google Drive API" 검색 후 사용 설정

### 2. 서비스 계정 생성
1. API 및 서비스 → 사용자 인증 정보
2. 사용자 인증 정보 만들기 → 서비스 계정
3. 서비스 계정 이름 입력 (예: "test-automation")
4. 역할: 편집자
5. 완료 후 서비스 계정 클릭
6. 키 탭 → 키 추가 → JSON 선택
7. 다운로드된 JSON 파일을 `credentials.json`으로 저장

### 3. Google Sheets 공유
1. 테스트 결과를 기록할 Google Sheets 생성
2. 공유 버튼 클릭
3. `credentials.json`의 `client_email` 복사
4. 해당 이메일에 편집자 권한 부여

### 4. **스크린샷 자동화**
```python
@pytest.hookimpl(tryfirst=True, hookwrapper=True)
def pytest_runtest_makereport(item, call):
    outcome = yield
    if outcome.get_result().failed:
        driver = item.funcargs.get('driver')
        if driver:
            driver.save_screenshot(f'fail_{item.name}.png')
```
